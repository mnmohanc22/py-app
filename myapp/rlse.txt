/opt/
├── myapp-releases/             # all deployed versions live here
│   ├── v1.0.0/                 # older version (kept for rollback)
│   ├── v1.0.1/                 # previous version
│   └── v1.1.0/                 # current version
│
├── myapp -> myapp-releases/v1.1.0   # symlink points to current
│
├── venvs/
│   └── app1/
├── envs/
│   └── app1.env
└── scripts/
    └── app1.sh

Developer
    │
    git tag -a v1.1.0 -m "Release"
    git push origin v1.1.0
    │
    ▼
CI/CD triggers
    │
    ansible-playbook deploy.yml -e git_version=v1.1.0
    │
    ▼
Ansible on server
    │
    ├── clone v1.1.0 → /opt/myapp-releases/v1.1.0
    ├── pip install  → /opt/venvs/app1  (unchanged if no new deps)
    ├── ln -sfn v1.1.0 → /opt/myapp    (symlink switches)
    ├── app1.sh stop
    ├── app1.sh start
    └── health check passes ✓
    │
    ▼
/opt/myapp-releases/
    ├── v1.0.0/    ← available for rollback
    ├── v1.0.1/    ← available for rollback
    └── v1.1.0/    ← current (symlinked)
